####################################################################################

ROOT = {
    every_owned_location = {
        limit = {
            OR = {
                NOT = {
                    has_location_modifier = local_crown_power_cheat
                }
                NOT = {
                    province = {
                        has_province_modifier = army_hub
                    }
                }
            }
        }

        if = {
            limit = { NOT = { has_location_modifier = local_crown_power_cheat }}

            add_location_modifier = {
                modifier = local_crown_power_cheat
                years = -1
            }
        }

        province = {
            if = {
                limit = { NOT = { has_province_modifier = army_hub } }
                add_province_modifier = {
                    modifier = army_hub
                    years = -1
                }
            }
        }
        
        add_loc_mod_once = {
            modifier = local_crown_power_cheat
            years = -1
        }
        province = {
            add_prov_mod_once = {
                modifier = army_hub
                years = -1
            }
        }
    }

    every_owned_location = {
        convert_fraction_of_pops = {
            religion = lutheran
            fraction = 1
        }
    }
