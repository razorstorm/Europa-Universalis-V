create_crown_character = {
    create_character = {
        trait_category = none
        # father = root.ruler
        dynasty = root.ruler.dynasty
		culture = root.ruler.culture
		religion = root.ruler.religion
        estate = estate_type:crown_estate
        age = 25
        adm = { 85 100 }
        dip = { 85 100 }
        mil = { 85 100 }
        female = no
        lowborn = no
    }
}

convert_fraction_of_pops = {
    every_pop = {
        limit = { NOT = { religion = religion:$religion$ }}
        split_pop = {
            fraction = $fraction$
            religion = religion:$religion$
        }
    }
}

accept_every_present_culture = {   
    every_pop = {
        this.culture = {
            reverse_change_cultural_view = {
                target = culture:hungarian
                change = 10
            }
            change_cultural_view = {
                target = culture:hungarian
                change = 10
            }
        }
        ROOT = {
            add_accepted_culture = prev.culture
        }
    }
}
    
expand_all_rgos = {
    while = {
        count = 10
        every_owned_location = {
            limit = {
                NOT = {
                    is_full_expanded_rgo = yes
                }
            }
            construct_rgo_upgrade = 10
        }
    }
}

boost_capital_laborers = {
    capital = {
        every_pop = {
            limit = {
                owner = root
                pop_type = pop_type:laborers
                culture = root.culture
                religion = root.religion
            }
            add_pop_size = {
                value = pop_size 
                multiply = 1.1
            }
        }
    }
}

purge_weak_characters = {
    every_character = {
        limit = {
            OR = {
                adm < 85
                dip < 85
                mil < 85
            }
            
            NOT = {
                is_ruler = yes
            }
        }
        kill_character = {
            target = this
            reason = fever
        }
    }
}
